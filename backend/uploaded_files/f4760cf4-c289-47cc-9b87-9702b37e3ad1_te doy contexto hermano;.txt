// Obtener excepciones por mes y año
export const fetchExcepcionesPorMesAnio = (month, year, page, size, showActive) => {
  const endpoint = showActive
    ? /workingscheduleexceptions/paginated/bymonthyear/active
    : /workingscheduleexceptions/paginated/bymonthyear;
  return apiToken.get(${endpoint}?month=${month}&year=${year}&page=${page}&size=${size}&sort=startDate);
};

// Buscar excepciones por fecha específica
export const fetchExcepcionesPorFecha = (date) => {
  return apiToken.get(/workingscheduleexceptions/bydate?date=${date});
};

// Obtener horarios de trabajo (días laborales)
export const fetchHorariosTrabajoApi = () => {
  return apiToken.get('/workingschedule');
};

// Crear nueva excepción
export const createExcepcionApi = (data) => {
  return apiToken.post('/workingscheduleexceptions', data);
};

// Actualizar excepción existente
export const updateExcepcionApi = (id, data) => {
  return apiToken.put(/workingscheduleexceptions/${id}, data);
};

// Obtener una excepción por ID
export const fetchExcepcionPorIdApi = (id) => {
  return apiToken.get(/workingscheduleexceptions/${id});
};

import { yupResolver } from '@hookform/resolvers/yup';
import * as yup from 'yup';
import Swal from 'sweetalert2';
import { FontAwesomeIcon } from '@fortawesome/react-fontawesome';
import { faClock, faCalendarAlt } from '@fortawesome/free-solid-svg-icons';
import CalendarioExcepciones from '../../../../components/admin/CalendarioExcepciones';

// Esquema de validación
const excepcionSchema = yup.object().shape({
  isRange: yup.boolean(),
  startDate: yup.date().required('La fecha de inicio es requerida'),
  endDate: yup.date().when('isRange', {
    is: true,
    then: (schema) => schema.required('La fecha de fin es requerida'),
    otherwise: (schema) => schema.nullable()
  }),
  partialTime: yup.boolean(),
  startHour: yup.string().when('partialTime', {
    is: true,
    then: (schema) => schema.required('La hora de inicio es requerida'),
    otherwise: (schema) => schema.nullable()
  }),
  endHour: yup.string().when('partialTime', {
    is: true,
    then: (schema) => schema.required('La hora de fin es requerida'),
    otherwise: (schema) => schema.nullable()
  })
});

const ModalExcepcion = ({ show, onHide, excepcionData = null, onSuccess }) => {
  const [selectedStartDate, setSelectedStartDate] = useState(null);
  const [selectedEndDate, setSelectedEndDate] = useState(null);
  
  const { crearExcepcion, actualizarExcepcion, isSubmitting } = useExcepcionForm();
  const { horariosTrabajo } = useHorariosTrabajo();

  const {
    control,
    handleSubmit,
    setValue,
    watch,
    reset,
    clearErrors,
    formState: { errors }
  } = useForm({
    resolver: yupResolver(excepcionSchema),
    defaultValues: {
      isRange: false,
      partialTime: false,
      startDate: null,
      endDate: null,
      startHour: '',
      endHour: ''
    }
  });

  const watchIsRange = watch('isRange');
  const watchPartialTime = watch('partialTime');

  // Generar opciones de horas (6:00 a 21:00)
  const generateTimeOptions = () => {
    const options = [];
    for (let hour = 6; hour <= 21; hour++) {
      for (let minute = 0; minute < 60; minute += 30) {
        const timeString = `${hour.toString().padStart(2, '0')}:${minute.toString().padStart(2, '0')}`;
        options.push({
          value: timeString,
          label: timeString
        });
      }
    }
    return options;
  };

  const timeOptions = generateTimeOptions();

  // Efecto para cargar datos en modo edición
  useEffect(() => {
    if (excepcionData && show) {
      const startDate = excepcionData.startDate ? new Date(excepcionData.startDate) : null;
      const endDate = excepcionData.endDate ? new Date(excepcionData.endDate) : null;
      
      setSelectedStartDate(startDate);
      setSelectedEndDate(endDate);
      
      setValue('isRange', !excepcionData.oneDay);
      setValue('partialTime', excepcionData.partialTime || false);
      setValue('startDate', startDate);
      setValue('endDate', endDate);
      setValue('startHour', excepcionData.startHour || '');
      setValue('endHour', excepcionData.endHour || '');
    }
  }, [excepcionData, show, setValue]);

  // Limpiar formulario al cerrar
  useEffect(() => {
    if (!show) {
      reset();
      setSelectedStartDate(null);
      setSelectedEndDate(null);
      clearErrors();
    }
  }, [show, reset, clearErrors]);

  // Manejar selección de fechas desde el calendario
  const handleDateSelect = (startDate, endDate = null) => {
    setSelectedStartDate(startDate);
    setSelectedEndDate(endDate);
    setValue('startDate', startDate);
    setValue('endDate', endDate);
  };

  // Formatear fecha para envío
  const formatDateForAPI = (date) => {
    if (!date) return null;
    const year = date.getFullYear();
    const month = (date.getMonth() + 1).toString().padStart(2, '0');
    const day = date.getDate().toString().padStart(2, '0');
    return `${year}-${month}-${day}`;
  };

  // Enviar formulario
  const onSubmit = async (data) => {
    try {
      const payload = {
        startDate: formatDateForAPI(data.startDate),
        oneDay: !data.isRange,
        endDate: data.isRange ? formatDateForAPI(data.endDate) : formatDateForAPI(data.startDate),
        partialTime: data.partialTime,
        startHour: data.partialTime ? data.startHour : null,
        endHour: data.partialTime ? data.endHour : null
      };

      let result;
      if (excepcionData?.id) {
        result = await actualizarExcepcion(excepcionData.id, payload);
      } else {
        result = await crearExcepcion(payload);
      }

      if (result.success) {
        Swal.fire({
          icon: 'success',
          title: excepcionData ? '¡Excepción Actualizada!' : '¡Excepción Registrada!',
          text: excepcionData 
            ? 'La excepción se ha actualizado correctamente'
            : 'La excepción se ha registrado correctamente',
          customClass: {
            confirmButton: 'confirm_btn'
          }
        }).then(() => {
          onHide();
          if (onSuccess) onSuccess();
        });
      } else {
        throw result.error;
      }
    } catch (error) {
      const errors = error.response?.data?.data;
      
      Swal.fire({
        icon: 'error',
        title: excepcionData ? '¡Error al actualizar!' : '¡Error al registrar!',
        text: !errors ? getAxiosErrorMessage(error) : undefined,
        html: errors
          ? `<ul style="text-align: left;">${Object.values(errors)
              .map((m) => `<li>${m}</li>`)
              .join('')}</ul>`
          : undefined,
        confirmButtonText: 'Aceptar',
        allowOutsideClick: false,
        allowEscapeKey: false,
        customClass: {
          confirmButton: 'confirm_btn'
        }
      });
    }
  };

  return (
    <Modal show={show} onHide={onHide} size="lg" backdrop="static">
      <Modal.Header closeButton>
        <Modal.Title>
          <FontAwesomeIcon icon={faCalendarAlt} className="me-2" />
          {excepcionData ? 'Editar Excepción de Atención' : 'Registro de Excepción de Atención'}
        </Modal.Title>
      </Modal.Header>

      <form onSubmit={handleSubmit(onSubmit)}>
        <Modal.Body>
          {/* Switch para rango de fechas */}
          <Row className="mb-4">
            <Col>
              <div className="range-switch-container">
                <span className="range-label">¿Es un rango de fechas (no se trabajará varios días seguidos)?</span>
                <Controller
                  name="isRange"
                  control={control}
                  render={({ field }) => (
                    <Form.Check
                      type="switch"
                      id="range-switch"
                      checked={field.value}
                      onChange={field.onChange}
                      className="custom-switch"
                    />
                  )}
                />
              </div>
            </Col>
          </Row>

          {/* Sección de fechas */}
          <Row className="mb-4">
            <Col>
              <div className="date-section">
                <h6 className="section-title">
                  {watchIsRange ? 'Selecciona un rango de fecha' : 'Selecciona una fecha'}
                </h6>
                
                {/* Inputs de fecha */}
                <div className="date-inputs">
                  <div className="date-input-group">
                    <label>DESDE</label>
                    <div className="date-display">
                      {selectedStartDate 
                        ? selectedStartDate.toLocaleDateString('es-ES', { 
                            day: 'numeric', 
                            month: 'long', 
                            year: 'numeric' 
                          })
                        : '24 abril | 2025'
                      }
                    </div>
                  </div>
                  
                  {watchIsRange && (
                    <div className="date-input-group">
                      <label>HASTA</label>
                      <div className="date-display">
                        {selectedEndDate 
                          ? selectedEndDate.toLocaleDateString('es-ES', { 
                              day: 'numeric', 
                              month: 'long', 
                              year: 'numeric' 
                            })
                          : '1 Mayo | 2025'
                        }
                      </div>
                    </div>
                  )}
                </div>

                {/* Calendario */}
                <CalendarioExcepciones
                  onDateSelect={handleDateSelect}
                  selectedStartDate={selectedStartDate}
                  selectedEndDate={selectedEndDate}
                  isRangeMode={watchIsRange}
                  horariosTrabajo={horariosTrabajo}
                />
                
                {errors.startDate && (
                  <div className="text-danger mt-2">{errors.startDate.message}</div>
                )}
                {errors.endDate && (
                  <div className="text-danger mt-2">{errors.endDate.message}</div>
                )}
              </div>
            </Col>
          </Row>

          {/* Switch para tiempo parcial */}
          <Row className="mb-4">
            <Col>
              <div className="partial-time-switch">
                <span className="switch-label">¿Se trabajará a tiempo parcial?</span>
                <Controller

          </Row>

          {/* Sección de horarios (solo si tiempo parcial está activado) */}
          {watchPartialTime && (
            <Row className="mb-4">

import React, { useState, useEffect } from 'react';
import { FontAwesomeIcon } from '@fortawesome/react-fontawesome';
import { faChevronLeft, faChevronRight } from '@fortawesome/free-solid-svg-icons';
import '../../css/admin/CalendarioExcepciones.css';
const CalendarioExcepciones = ({ 
  onDateSelect, 
  selectedStartDate, 
  selectedEndDate, 
  isRangeMode,
  horariosTrabajo = []
}) => {
  const [currentMonth, setCurrentMonth] = useState(new Date().getMonth());
  const [currentYear, setCurrentYear] = useState(new Date().getFullYear());
  const [secondMonth, setSecondMonth] = useState(new Date().getMonth() + 1);
  const [secondYear, setSecondYear] = useState(new Date().getFullYear());
  // Mapeo de días de la semana para comparar con la API
  const diasSemanaMap = {
    0: 'DOMINGO',
    1: 'LUNES', 
    2: 'MARTES',
    3: 'MIERCOLES',
    4: 'JUEVES',
    5: 'VIERNES',
    6: 'SABADO'
  };
  // Obtener días laborales de la API
  const diasLaborales = horariosTrabajo.map(h => h.dayOfWeek);
  // Verificar si un día es laborable
  const esDiaLaborable = (fecha) => {
    const diaSemana = diasSemanaMap[fecha.getDay()];
    return diasLaborales.includes(diaSemana);
  };
  useEffect(() => {
    if (isRangeMode) {
      let nextMonth = currentMonth + 1;
      let nextYear = currentYear;
      if (nextMonth > 11) {
        nextMonth = 0;
        nextYear = currentYear + 1;
      }
      setSecondMonth(nextMonth);
      setSecondYear(nextYear);
    }
  }, [currentMonth, currentYear, isRangeMode]);
  const meses = [
    'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio',
    'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'
  ];
  const diasSemana = ['Lun', 'Mar', 'Mié', 'Jue', 'Vie', 'Sáb', 'Dom'];
  const getDaysInMonth = (month, year) => {
    return new Date(year, month + 1, 0).getDate();
  };
  const getFirstDayOfMonth = (month, year) => {
    const firstDay = new Date(year, month, 1).getDay();
    return firstDay === 0 ? 6 : firstDay - 1; // Ajustar para que lunes sea 0
  };
  const navigateMonth = (direction, isSecondCalendar = false) => {
    if (isSecondCalendar) {
      let newMonth = direction === 'prev' ? secondMonth - 1 : secondMonth + 1;
      let newYear = secondYear;

      if (newMonth < 0) {
        newMonth = 11;
        newYear -= 1;
      } else if (newMonth > 11) {
        newMonth = 0;
        newYear += 1;
      }

      setSecondMonth(newMonth);
      setSecondYear(newYear);
    } else {
      let newMonth = direction === 'prev' ? currentMonth - 1 : currentMonth + 1;
      let newYear = currentYear;

      if (newMonth < 0) {
        newMonth = 11;
        newYear -= 1;
      } else if (newMonth > 11) {
        newMonth = 0;
        newYear += 1;
      }

      setCurrentMonth(newMonth);
      setCurrentYear(newYear);
    }
  };
  const handleDateClick = (day, month, year) => {
    const clickedDate = new Date(year, month, day);

    // Verificar si es día laborable
    if (!esDiaLaborable(clickedDate)) {
      return; // No permitir selección de días no laborables
    }
    if (isRangeMode) {
      if (!selectedStartDate || (selectedStartDate && selectedEndDate)) {
        // Seleccionar fecha de inicio o reiniciar selección
        onDateSelect(clickedDate, null);
      } else {
        // Seleccionar fecha de fin
        if (clickedDate >= selectedStartDate) {
          onDateSelect(selectedStartDate, clickedDate);
        } else {
          onDateSelect(clickedDate, null);
        }
      }
    } else {
      onDateSelect(clickedDate);
    }
  };
  const isDateSelected = (day, month, year) => {
    const date = new Date(year, month, day);
    if (isRangeMode && selectedStartDate && selectedEndDate) {
      return date >= selectedStartDate && date <= selectedEndDate;
    }
    if (selectedStartDate) {
      return date.toDateString() === selectedStartDate.toDateString();
    }
    return false;
  };
  const isStartDate = (day, month, year) => {
    if (!selectedStartDate) return false;
    const date = new Date(year, month, day);
    return date.toDateString() === selectedStartDate.toDateString();
  };
  const isEndDate = (day, month, year) => {
    if (!selectedEndDate) return false;
    const date = new Date(year, month, day);
    return date.toDateString() === selectedEndDate.toDateString();
  };
  const renderCalendar = (month, year, isSecond = false) => {
    const daysInMonth = getDaysInMonth(month, year);
    const firstDay = getFirstDayOfMonth(month, year);
    const days = [];
    // Días vacíos al inicio
    for (let i = 0; i < firstDay; i++) {
      days.push(<div key={empty-${i}} className="calendar-day empty"></div>);
    }
    // Días del mes
    for (let day = 1; day <= daysInMonth; day++) {
      const fecha = new Date(year, month, day);
      const isLaborable = esDiaLaborable(fecha);
      const isSelected = isDateSelected(day, month, year);
      const isStart = isStartDate(day, month, year);
      const isEnd = isEndDate(day, month, year);
      let className = 'calendar-day';
      if (!isLaborable) className += ' disabled';
      if (isSelected) className += ' selected';
      if (isStart) className += ' start-date';
      if (isEnd) className += ' end-date';
      days.push(
        <div
          key={day}
          className={className}
          onClick={() => isLaborable && handleDateClick(day, month, year)}
        >
          {day}
        </div>
      );
    }
    return (
      <div className="calendar-container">
        <div className="calendar-header">
          <button 
            type="button"
            className="nav-button" 
            onClick={() => navigateMonth('prev', isSecond)}
          >
            <FontAwesomeIcon icon={faChevronLeft} />
          </button>
          <h4>{meses[month]} {year}</h4>
          <button 
            type="button"
            className="nav-button" 
            onClick={() => navigateMonth('next', isSecond)}
          >
            <FontAwesomeIcon icon={faChevronRight} />
          </button>
        </div>

        <div className="calendar-weekdays">
          {diasSemana.map(dia => (
            <div key={dia} className="weekday">{dia}</div>
          ))}
        </div>

        <div className="calendar-grid">
          {days}
        </div>
      </div>
    );
  };
  return (
    <div className="calendario-excepciones">
      {isRangeMode ? (
        <div className="dual-calendar">
          {renderCalendar(currentMonth, currentYear)}
          {renderCalendar(secondMonth, secondYear, true)}
        </div>
      ) : (
        <div className="single-calendar">
          {renderCalendar(currentMonth, currentYear)}
        </div>
      )}
    </div>
  );
};
export default CalendarioExcepciones;

  { value: 6, label: 'Junio' },
  { value: 7, label: 'Julio' },
  { value: 8, label: 'Agosto' },
  { value: 9, label: 'Septiembre' },
  { value: 10, label: 'Octubre' },
  { value: 11, label: 'Noviembre' },
  { value: 12, label: 'Diciembre' }
];

const AdminExcepcionesHorarios = () => {
  const today = new Date();
  const [selectedMonth, setSelectedMonth] = useState(today.getMonth() + 1);
  const [selectedYear, setSelectedYear] = useState(today.getFullYear());
  const [showMonthYearModal, setShowMonthYearModal] = useState(false);
  const [showExcepcionModal, setShowExcepcionModal] = useState(false);
  const [currentPage, setCurrentPage] = useState(0);
  const [showActiveOnly, setShowActiveOnly] = useState(false);
  const [searchTerm, setSearchTerm] = useState('');
  const [searchDate, setSearchDate] = useState('');
  const [excepcionEditando, setExcepcionEditando] = useState(null);
  const [refrescarDatos, setRefrescarDatos] = useState(0);

  const { exceptions, loading, totalPages } = useExcepcionesAtencion(
    selectedMonth,
    selectedYear,
    currentPage,
    showActiveOnly,
    searchDate
  );

  const handleSearchTerm = (term) => {
    setSearchTerm(term);
    setSearchDate(term || '');
    setCurrentPage(0);
  };

  const handleMonthYearChange = () => {
    setShowMonthYearModal(false);
    setCurrentPage(0);
    setSearchDate('');
    setSearchTerm('');
  };

  // Abrir modal para nueva excepción
  const handleNuevaExcepcion = () => {
    setExcepcionEditando(null);
    setShowExcepcionModal(true);
  };

  // Abrir modal para editar excepción
  const handleEditarExcepcion = (excepcion) => {
    setExcepcionEditando(excepcion);
    setShowExcepcionModal(true);
  };

  // Cerrar modal de excepción
  const handleCerrarModal = () => {
    setShowExcepcionModal(false);
    setExcepcionEditando(null);
  };

  // Éxito al guardar excepción
  const handleExitoExcepcion = () => {
    setRefrescarDatos(prev => prev + 1); // Refrescar datos
  };

  // Formatear fecha sin desfase horario
  const formatDate = (dateString) => {
    if (!dateString) return 'No laborable';
    // Parseamos directamente la fecha como está, sin agregar zona horaria
    const [year, month, day] = dateString.split('-');
    const date = new Date(year, month - 1, day);
    return date.toLocaleDateString('es-ES', { 
      day: 'numeric', 
      month: 'long', 
      year: 'numeric' 
    });
  };

  const formatTime = (timeString) => timeString || 'No laborable';

  // Cambiar página
  const handlePageChange = (page) => setCurrentPage(page);

  // Para mostrar nombre del mes en título
  const currentMonthName = months.find(m => m.value === selectedMonth)?.label || '';

  return (
    <div className="admin-excepciones-horarios">
      <div className="header-section">
        <h2>Excepciones de atención</h2>
      </div>

      <SearchBar
        searchTerm={searchTerm}
        setSearchTerm={handleSearchTerm}
        placeholder="Buscar fecha"
        type="date"
      >
        <Form.Check
          type="switch"
          id="active-only-switch"
          checked={showActiveOnly}
          label="Ver solo excepciones activas"
          onChange={(e) => { setShowActiveOnly(e.target.checked); setCurrentPage(0); }}
        />
        <button className="boton" onClick={handleNuevaExcepcion}>
          <FontAwesomeIcon icon={faPlus} />
          Nuevo
        </button>
      </SearchBar>

      {/* Título y selector de mes/año fuera de la tabla */}
      <div className="table-header">
        <div className="month-year-title">
          <h5>Excepciones {currentMonthName} - {selectedYear}</h5>
          <button
            className="month-year-selector-btn"
            onClick={() => setShowMonthYearModal(true)}
            title="Cambiar mes y año"
          >
            {currentMonthName.toUpperCase()}-{selectedYear}
          </button>
        </div>
      </div>

      <div className="table-container">
        <Table responsive className="exceptions-table">
          <thead>
            <tr>
              <th>Fecha</th>
              <th>Horario de atención</th>
              <th>Estatus</th>
              <th>Acciones</th>
            </tr>
          </thead>
          <tbody>
            {loading ? (
              <tr><td colSpan="4" className="text-center">Cargando...</td></tr>
            ) : exceptions.length === 0 ? (
              <tr><td colSpan="4" className="text-center">No se encontraron excepciones</td></tr>
            ) : (
              exceptions.map((exception, index) => (
                <tr key={exception.id || index}>
                  <td>
                    <div className="date-info">
                      <div className="date-text">
                        {formatDate(exception.startDate)}
                        {exception.endDate && exception.endDate !== exception.startDate && ` - ${formatDate(exception.endDate)}`}
                      </div>
                      <div className="date-subtitle">
                        {exception.partialTime 
                          ? 'Tiempo parcial' 
                          : (exception.description || 'No laborable')
                        }
                      </div>
                    </div>
                  </td>
                  <td>
                    {exception.partialTime
                      ? `${formatTime(exception.startHour)} - ${formatTime(exception.endHour)}`
                      : (exception.startHour && exception.endHour)
                        ? `${formatTime(exception.startHour)} - ${formatTime(exception.endHour)}`
                        : 'No laborable'
                    }
                  </td>
                  <td>
                    <div className="status-container">
                      <span className={`status-indicator ${exception.status ? 'active' : 'inactive'}`}></span>
                      <span className="status-text">
                        {exception.status ? 'Activo' : 'Inactivo'}
                      </span>
                    </div>
                  </td>
                  <td>
                    <div className="actions-column">
                      <button 
                        className="edit_button"
                        onClick={() => handleEditarExcepcion(exception)}
                      >
                        <FontAwesomeIcon icon={faEdit} />
                        Editar
                      </button>
                      <Form.Check
                        type="switch"
                        id={`switch-${exception.id || index}`}
                        checked={exception.status || false}
                        onChange={() => console.log(`Toggle status de excepción id: ${exception.id}`)} 
                      />
                    </div>
                  </td>
                </tr>
              ))
            )}
          </tbody>
        </Table>

        <Pagination
          currentPage={currentPage}
          totalPages={totalPages}
          onPageChange={handlePageChange}
        />
      </div>

      {/* Modal para seleccionar mes y año */}
      <ModalMesAnio
        show={showMonthYearModal}
        onHide={() => setShowMonthYearModal(false)}
        selectedMonth={selectedMonth}
        setSelectedMonth={setSelectedMonth}
        selectedYear={selectedYear}
        setSelectedYear={setSelectedYear}
        onApply={handleMonthYearChange}
      />

      {/* Modal para crear/editar excepción */}
      <ModalExcepcion
        show={showExcepcionModal}
        onHide={handleCerrarModal}
        excepcionData={excepcionEditando}
        onSuccess={handleExitoExcepcion}
        key={refrescarDatos} // Forzar re-render cuando se actualicen los datos
      />
    </div>
  );
};

export default AdminExcepcionesHorarios;
